## Step‑by‑step: Python script to create 2 combined CSVs

This script will:

- Download all five indicators.  
- Keep only the essential fields (`REF_AREA`, `TIME_PERIOD`, `INDICATOR`, `OBS_VALUE`).  
- Pivot them so each **indicator becomes a column**.  
- Split by country and save:
  - `senegal_gender_dataset_2019_2024.csv`  
  - `cote_d_ivoire_gender_dataset_2019_2024.csv`

```python
import requests
import pandas as pd

# 1. Define the indicators and URLs
indicator_urls = {
    "CPIA_gender_equality": "https://data360api.worldbank.org/data360/data?DATABASE_ID=WB_CPIA&INDICATOR=WB_CPIA_IQ_CPA_GNDR_XQ&REF_AREA=SEN,CIV&timePeriodFrom=2019&timePeriodTo=2024&top=2000",
    "Gender_Inequality_Index": "https://data360api.worldbank.org/data360/data?DATABASE_ID=FAO_AS&INDICATOR=FAO_AS_4537&REF_AREA=SEN,CIV&timePeriodFrom=2019&timePeriodTo=2024&top=2000",
    "Female_LFPR_15plus": "https://data360api.worldbank.org/data360/data?DATABASE_ID=WB_WDI&INDICATOR=WB_WDI_SL_TLF_CACT_FE_ZS&REF_AREA=SEN,CIV&timePeriodFrom=2019&timePeriodTo=2024&top=2000",
    "Women_in_parliament": "https://data360api.worldbank.org/data360/data?DATABASE_ID=WB_WDI&INDICATOR=WB_WDI_SG_GEN_PARL_ZS&REF_AREA=SEN,CIV&timePeriodFrom=2019&timePeriodTo=2024&top=2000",
    "Girls_lower_secondary_completion": "https://data360api.worldbank.org/data360/data?DATABASE_ID=WB_WDI&INDICATOR=WB_WDI_SE_SEC_CMPT_LO_FE_ZS&REF_AREA=SEN,CIV&timePeriodFrom=2019&timePeriodTo=2024&top=2000",
}

# 2. Helper function to fetch and tidy one indicator
def fetch_indicator(name, url):
    r = requests.get(url)
    r.raise_for_status()
    raw = r.json()

    # Data are under the "data" key
    df = pd.DataFrame(raw["data"])

    # Keep only core fields
    df = df[["REF_AREA", "TIME_PERIOD", "OBS_VALUE"]].copy()

    # Convert OBS_VALUE to numeric if possible
    df["OBS_VALUE"] = pd.to_numeric(df["OBS_VALUE"], errors="coerce")

    # Rename OBS_VALUE to the indicator's friendly name
    df = df.rename(columns={"OBS_VALUE": name})

    return df

# 3. Fetch all indicators and merge them
dfs = []

for name, url in indicator_urls.items():
    print(f"Fetching {name} ...")
    df_i = fetch_indicator(name, url)
    dfs.append(df_i)

# Start from the first indicator and merge others on REF_AREA + TIME_PERIOD
from functools import reduce

combined = reduce(
    lambda left, right: pd.merge(
        left, right, on=["REF_AREA", "TIME_PERIOD"], how="outer"
    ),
    dfs,
)

# 4. Sort by country and year for readability
combined["TIME_PERIOD"] = combined["TIME_PERIOD"].astype(int)
combined = combined.sort_values(["REF_AREA", "TIME_PERIOD"])

# 5. Split by country and save one CSV per country
# REF_AREA codes: SEN = Senegal, CIV = Côte d'Ivoire
senegal = combined[combined["REF_AREA"] == "SEN"].copy()
cote_divoire = combined[combined["REF_AREA"] == "CIV"].copy()

senegal.to_csv("senegal_gender_dataset_2019_2024.csv", index=False)
cote_divoire.to_csv("cote_d_ivoire_gender_dataset_2019_2024.csv", index=False)

print("Saved:")
print("  senegal_gender_dataset_2019_2024.csv")
print("  cote_d_ivoire_gender_dataset_2019_2024.csv")
```

Each CSV will have rows by **year** and columns:
