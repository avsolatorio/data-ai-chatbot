  - Use of cohort component methods and projections.

- **Update timing / vintages** (HNP vs WDI may not be synchronized to the same UN WPP release at all times).

```python
import requests

BASE_URL = "https://data360api.worldbank.org/data360/data"
MICROSTATES = ["AND", "MCO", "LIE", "SMR", "MLT"]  # extend as needed
START_YEAR = 2015
END_YEAR = 2024
WARN_THRESHOLD = 0.02  # 2%
ERROR_THRESHOLD = 0.05  # 5%

def fetch_series(database_id, indicator_id, country):
    params = {
        "DATABASE_ID": database_id,
        "INDICATOR": indicator_id,
        "REF_AREA": country,
        "TIME_PERIOD": f"{START_YEAR}:{END_YEAR}",
    }
    r = requests.get(BASE_URL, params=params)
    r.raise_for_status()
    js = r.json()
    if "data" not in js or not js["data"]:
        return {}
    return {
        int(d["TIME_PERIOD"]): float(d["OBS_VALUE"])
        for d in js["data"]
        if d.get("OBS_VALUE") is not None
    }

def test_microstate_population_consistency():
    failures = []
    warnings = []

    for c in MICROSTATES:
        hnp = fetch_series("WB_HNP", "WB_HNP_SP_POP_TOTL", c)
        wdi = fetch_series("WB_WDI", "WB_WDI_SP_POP_TOTL", c)

        common_years = sorted(set(hnp) & set(wdi))
        if not common_years:
            warnings.append((c, "no_common_years", None))
            continue

        for year in common_years:
            v_hnp = hnp[year]
            v_wdi = wdi[year]
            diff_abs = abs(v_hnp - v_wdi)
            denom = max(v_hnp, v_wdi)
            if denom == 0:
                continue
            diff_rel = diff_abs / denom

            if diff_rel >= ERROR_THRESHOLD:
                failures.append((c, year, v_hnp, v_wdi, diff_rel))
            elif diff_rel >= WARN_THRESHOLD:
                warnings.append((c, year, v_hnp, v_wdi, diff_rel))

    # You can assert on failures to make the test fail:
    assert not failures, f"Large population discrepancies: {failures}"

    # Optionally log warnings for inspection
    if warnings:
        print("Population warnings (microstates):")
        for w in warnings:
            print(w)
```

import requests

BASE_URL = "https://data360api.worldbank.org/data360/data"
MICROSTATES = ["AND", "MCO", "LIE", "SMR", "MLT"]  # extend as needed
START_YEAR = 2015
END_YEAR = 2024
WARN_THRESHOLD = 0.02  # 2%
ERROR_THRESHOLD = 0.05  # 5%

def fetch_series(database_id, indicator_id, country):
    params = {
        "DATABASE_ID": database_id,
        "INDICATOR": indicator_id,
        "REF_AREA": country,
        "TIME_PERIOD": f"{START_YEAR}:{END_YEAR}",
    }
    r = requests.get(BASE_URL, params=params)
    r.raise_for_status()
    js = r.json()
    if "data" not in js or not js["data"]:
        return {}
    return {
        int(d["TIME_PERIOD"]): float(d["OBS_VALUE"])
        for d in js["data"]
        if d.get("OBS_VALUE") is not None
    }

def test_microstate_population_consistency_with_coverage():
    value_failures = []   # large relative differences
    value_warnings = []   # moderate relative differences
    coverage_gaps = []    # years present in one DB but not the other

    for c in MICROSTATES:
        hnp = fetch_series("WB_HNP", "WB_HNP_SP_POP_TOTL", c)
        wdi = fetch_series("WB_WDI", "WB_WDI_SP_POP_TOTL", c)

        years_hnp = set(hnp.keys())
        years_wdi = set(wdi.keys())

        common_years = sorted(years_hnp & years_wdi)
        only_hnp = sorted(years_hnp - years_wdi)
        only_wdi = sorted(years_wdi - years_hnp)

        # --- coverage gaps ---
        if only_hnp:
            coverage_gaps.append({
                "country": c,
                "missing_in": "WB_WDI",
                "years": only_hnp,
            })
        if only_wdi:
            coverage_gaps.append({
                "country": c,
                "missing_in": "WB_HNP",
                "years": only_wdi,
            })

        # If there are no overlapping years, skip value comparison
        if not common_years:
            continue

        # --- value discrepancies on common years ---
        for year in common_years:
            v_hnp = hnp[year]
            v_wdi = wdi[year]
            diff_abs = abs(v_hnp - v_wdi)
            denom = max(v_hnp, v_wdi)
            if denom == 0:
                continue
            diff_rel = diff_abs / denom

            record = {
                "country": c,
                "year": year,
                "hnp": v_hnp,
                "wdi": v_wdi,
                "diff_abs": diff_abs,
                "diff_rel": diff_rel,
            }

            if diff_rel >= ERROR_THRESHOLD:
                value_failures.append(record)
            elif diff_rel >= WARN_THRESHOLD:
                value_warnings.append(record)

    # Decide what should fail the test:
    # 1) large value discrepancies
    # 2) optionally, coverage gaps in recent years
    assert not value_failures, f"Large population discrepancies: {value_failures}"

    # If you want coverage gaps to also fail, uncomment:

        print("Population value warnings (microstates):")
        for w in value_warnings:
            print(w)

    if coverage_gaps:
        print("Population coverage gaps (microstates):")
        for g in coverage_gaps:
            print(g)
